Datatype DaWVZStellzustandSend	ohne Zeit Reaktion Write Area
  Fill Low (anzeigeprinzip)
  Fill Low (stellcode)
  If (funktionsbyte == 1 && anzeigeprinzip == 1 && (stellcode == 61 || stellcode == 113 || stellcode == 114)) {
    Fill Low (82)
  } Else {
  If ($LogischeAdresse == 32938-71 || $LogischeAdresse == 32939-0) {
	If ($DeNummer == 113 && funktionsbyte == 1 && (stellcode == 202 || stellcode == 203)) {
		Fill Low (162)
	} Else {
		Fill Low (funktionsbyte)
	}
  } Else {
		Fill Low (funktionsbyte)
	}
  }
  If (anzahl > 0 && stellcode == 0) {
    Fill Low (anzahl)
    If (anzeigeprinzip == 4) {
  	  IncLoop(i, 0, anzahl) {
        Fill Low	(code[$i*2])
        Fill Low	(code[$i*2+1])
  	  }
    } Else {
	  Fill Block code Offset 0 Size anzahl
	}
  }
