/* ========================================================================== */
/* Converts incoming SYSKommunikationsstatus messages to SYSFehlerDUE         */
/* messages with manufacturer HeuBoe.                                         */
/* ========================================================================== */
Bei Eingang von SYSKommunikationsstatus
{
    SYSFehlerDUE.fehlercode := status
    SYSFehlerDUE.hersteller := 6
}

/* ========================================================================== */
/* Temporary helper for due2de.                                               */
/* ========================================================================== */
Bei Eingang von LVEDeFehler (self=true)
{
    if ((fehlercode == 128) && (hersteller == 6)) {
        DEs des Knoten fg(1) in in.LVEDeFehler.fehlercode := 0
        DEs des Knoten fg(1) in in.LVEDeFehler.hersteller := 6
    }
}
