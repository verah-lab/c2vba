			Ea(2025559) in DaSYSSteuerSequenz.Aktion	:= LOKALE_ZEITSYNCHRONISATION
		}
		
		/* 2027841 KRI_A02_SH-Graz-West (KRI-device) ->  2025566 KRI_A02_SH-Graz-West(Q-KRI) */    
		If ($Eaid == 2027841) {
			Ea(2025566) in DaSYSSteuerSequenz.Aktion	:= LOKALE_ZEITSYNCHRONISATION
		}
		
		/* 2027842 KRI_Pack (KRI-device) ->  2025572 KRI_Pack (Q-KRI) */    
		If ($Eaid == 2027842) {
			Ea(2025572) in DaSYSSteuerSequenz.Aktion	:= LOKALE_ZEITSYNCHRONISATION
		}
		
		/* 2037356 KRI_2B_A02_SH_SBA_GRAZ_KN_WEST -> 2036668 KRI_2B_A02_SH_SBA_GRAZ_KN_WEST */    
		If ($Eaid == 2037356) {
			Ea(2036668) in DaSYSSteuerSequenz.Aktion	:= LOKALE_ZEITSYNCHRONISATION
		}

		If ($Eaid == 2037357) {
			Ea(2036673) in DaSYSSteuerSequenz.Aktion	:= LOKALE_ZEITSYNCHRONISATION
		}
		
	}
}
