/*******************************************************************************
*
*	GetValue[1-4]* Konvention
*		Die Anzahl der Ziffern gibt an, wieviele Bytes gelesen werden
*		Die Ziffer gibt an, welche Wertigkeit das jeweile Byte in der
*		Zielvariablen einnehmen soll. 1 Gibt hierbei das
*		niederwertigste Byte an.
*
*	Beispiel:
*		GetValue3412
*			1. gelesene Byte -> 3. Byte im Ergebnis (vHigh)
*			2. gelesene Byte -> 4. Byte im Ergebnis (eHigh)
*			3. gelesene Byte -> 1. Byte im Ergebnis (Low)
*			4. gelesene Byte -> 2. Byte im Ergebnis (High)
*
*	Damit wird aus
*		1.	2.	3.	4.	eHigh	vHigh	High	Low
*		0xaa	0xbb	0xcc	0xdd ->	0xbb	0xaa	0xdd	0xcc
*
*******************************************************************************/
#ifndef GetValue21
#define GetValue21(x) Get High (x) Get Low (x) 
#endif

#ifndef GetValue4321
#define GetValue4321(x) Get eHigh (x) Get vHigh (x) Get High (x) Get Low (x) 
#endif

#ifndef GetAndAssign
#define GetAndAssign(x) Get Low ($ x) x := $ x
#endif
