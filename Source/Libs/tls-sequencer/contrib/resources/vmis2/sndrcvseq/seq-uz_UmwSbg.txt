/* Externe Trigger aufgrund von Roten und Gelben LSA vor Tunnelportalen
*/
Bei Eingang von DaVLTHoehenkontrolle_erf
{
  /* VLT_Q_HOEHENKONTROLLE        ("HK_A01_1_941~HK", "HK_A01_1_284,204~HK", 32911 28, 4, 
     2056002|112|HK_A01_1_941~HK|9|Y|
	 2056005|112|TQ_Liefering_Muenchen_Hoehenk|9|Y|
  */
  If ($Eaid == 2056002) {
    If (DaVLTHoehenkontrolle_erf-New.zustand == 2) {
      Ea(2056005) in DaTriggerSchaltzustand.schaltzustand := 1
    } Else {
      Ea(2056005) in DaTriggerSchaltzustand.schaltzustand := 0
    }
  }

  /* VLT_Q_HOEHENKONTROLLE        ("HK_A01_2_982~HK", "HK_A01_2_296,500~HK", 32916 8, 4,
     2056003|112|HK_A01_2_982~HK|9|Y|
	 2056004|112|TQ_Liefering_Wien_Hoehenk|9|Y|
  */
  If ($Eaid == 2056003) {
    If (DaVLTHoehenkontrolle_erf-New.zustand == 2) {
      Ea(2056004) in DaTriggerSchaltzustand.schaltzustand := 1
    } Else {
      Ea(2056004) in DaTriggerSchaltzustand.schaltzustand := 0
    }
  }
  
}
