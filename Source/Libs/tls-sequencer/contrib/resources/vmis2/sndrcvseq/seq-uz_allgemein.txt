/* FG 1 (DE-Gut-Meldung)                                                      */
/* ========================================================================== */

Bei Eingang von DaLVEFehlerDE
{
  If ( (Fehlercode & DE_STOER_BITS) == DE_GUT_MELDUNG)
  {
    
    If ($DeNummer < 193 ) {
      DaLVESteuerSequenz.Aktion := LVE_DE_SENDEN_KANAL_STEUERUNG
	}
  }
}

Bei Eingang von DaLVESteuerSequenz
{
  If ($DeNummer < 193 )
  {
     Switch( Aktion ) {
       Case LVE_DE_SENDEN_KANAL_STEUERUNG:
            Aktion := LVE_DE_SENDEN_LVE_BETRIEBSPARAMETER
            Break
       Case LVE_DE_SENDEN_LVE_BETRIEBSPARAMETER:
            Aktion := LVE_DE_SENDEN_VKLASSE_LANG
            Break
       Case LVE_DE_SENDEN_VKLASSE_LANG:
            Aktion := LVE_DE_SENDEN_VKLASSE_KURZ
            Break
     }
  }
}

/* FG 4 (DE-Gut-Meldung)                                                      */
/* ========================================================================== */

Bei Eingang von DaWVZFehlerDEIst
{
  If ( (Fehlercode & DE_STOER_BITS) == DE_GUT_MELDUNG)
  {
    If ($DeNummer > 192 && $DeNummer < 224) {
       // Clusterkanal
      DaWVZSteuerSequenz.Aktion := WVZ_DE_SENDEN_BETRIEBSART
    } Else {
      // normales WZG
      DaWVZSteuerSequenz.Aktion := WVZ_DE_SENDEN_KANAL_STEUERUNG
    }
  }
}

Bei Eingang von DaWVZSteuerSequenz
{
  Switch(Aktion) {
    Case WVZ_DE_SENDEN_KANAL_STEUERUNG:
      Aktion := WVZ_DE_SENDEN_GRUNDEINSTELLUNG
      Break
  }
}


/* ====================================================================================== */
/* FG 4 (Meldung Autarkbetrieb)
/*   Einbaugrund: 
/*   Eine KRI2C versendet zur Zeit bei Ruecknahme einer externen Steuerung
/*   die Einstellung des Autarkbetriebs. (Die AQ sollen in die Grundeinstellung)
/*   Damit die Betriebsart "Autakrbetrieb" nicht laengere Zeit anliegt wird nun hier sofort
/*   die eigentlich eingestellte Betriebsart versendet. (siehe bug 1973)
/*
/*   Hinweis: Wird der Autarkbetrieb nach Übergang SSt tot -> lebt an die UZ versendet,
/*            so wird nun in diesem Fall die Betriebsart zweimal an den Cluster
/*            versendet.
/* ====================================================================================== */

Bei Eingang von DaWVZBetrArtPhysIst
{
	If ( WVZBetrArt == AUTARKBETRIEB )
	{
		DaWVZSteuerSequenz.Aktion	:=	WVZ_DE_SENDEN_BETRIEBSART
	}
}
