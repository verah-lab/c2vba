/* FG 1 (DE-Gut-Meldung)                                                      */
/* ========================================================================== */

Bei Eingang von LVEDeFehler (self=true)
{
    if (((fehlercode & DE_STOER_BITS) == DE_GUT_MELDUNG) && (($DeNummer <= 192) || ($DeNummer >= 224))) {
        SteuerSequenz.action via out.LVESteuerSequenz := LVE_DE_SENDEN_KANAL_STEUERUNG
        SteuerSequenz.fg via out.LVESteuerSequenz := 1
    }
}

Bei Eingang von SteuerSequenz via out.LVESteuerSequenz (self=true)
{
    if (action == LVE_DE_SENDEN_KANAL_STEUERUNG) {
        SteuerSequenz.action via out.LVESteuerSequenz := LVE_DE_SENDEN_LVE_BETRIEBSPARAMETER
        SteuerSequenz.fg via out.LVESteuerSequenz := 1
    }
}

/* ====================================================================================== */
/* FG 4 (Meldung Autarkbetrieb)
/*   Einbaugrund:
/*   Eine KRI2C versendet zur Zeit bei Ruecknahme einer externen Steuerung
/*   die Einstellung des Autarkbetriebs. (Die AQ sollen in die Grundeinstellung)
/*   Damit die Betriebsart "Autakrbetrieb" nicht laengere Zeit anliegt wird nun hier sofort
/*   die eigentlich eingestellte Betriebsart versendet. (siehe bug 1973)
/*
/*   Hinweis: Wird der Autarkbetrieb nach Übergang SSt tot -> lebt an die UZ versendet,
/*            so wird nun in diesem Fall die Betriebsart zweimal an den Cluster
/*            versendet.
/* ====================================================================================== */

Bei Eingang von WZGBetriebsart
{
	if (betriebsart == AUTARKBETRIEB)
	{
		SteuerSequenz.action via out.WZGSteuerSequenz := WZG_DE_SENDEN_BETRIEBSART
		SteuerSequenz.fg via out.WZGSteuerSequenz := 4
	}
}
